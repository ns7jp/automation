# =====================================================================
# backup_cron.txt
# ex10: cron定義を書く(解答例)
#
# 解答例は「1つの正解」ではありません。仕様を満たしていれば
# コメントの文面や順番は自由です。ただし「1行ごとに、何をいつ実行するかを
# 日本語で書き添える」書き方は実務でも定番なので、真似することを勧めます。
#
# 反映のしかた:
#   crontab backup_cron.txt     # このファイルの内容で丸ごと登録する
#   crontab -l                  # 登録内容を確認する
#
# 書式: 分 時 日 月 曜日 実行するコマンド
#       分(0-59) 時(0-23) 日(1-31) 月(1-12) 曜日(0-7、0と7は日曜)
#       「*」は「その項目を絞り込まない=毎回」を意味する
# =====================================================================

# cron が実行結果のメールを送る宛先。
# ジョブが出力(標準出力・標準エラー出力)を出すと、この宛先にメールが飛ぶ。
# 各行で出力をログへリダイレクトしているため、実際にメールが飛ぶのは
# リダイレクト自体が失敗するような異常時だけになる。
MAILTO="admin@example.com"

# 毎日 深夜2時30分 にバックアップを取得する。
# 実行結果は、標準出力も標準エラー出力もまとめて backup.log に追記する。
30 2 * * * /opt/scripts/backup.sh >> /var/log/backup.log 2>&1

# 平日(月曜から金曜)の 9時から18時まで、毎時0分 に死活監視を実行する。
# 曜日フィールドの 1-5 が月曜から金曜、時フィールドの 9-18 が9時から18時を表す。
0 9-18 * * 1-5 /opt/scripts/health_check.sh >> /var/log/health_check.log 2>&1

# 5分ごと にログ監視を実行する。
# 実行回数が多く、成功時の出力までログに残すと肥大化するため出力は捨てる。
# 検知した異常は log_watch.sh 自身が通知する設計にしておく。
*/5 * * * * /opt/scripts/log_watch.sh > /dev/null 2>&1

# 毎月1日の 3時0分 に、古いバックアップを7世代だけ残して削除する。
# 削除した内容はあとから確認できるよう rotate.log に追記する。
0 3 1 * * /opt/scripts/rotate_backups.sh -d /backup -g 7 >> /var/log/rotate.log 2>&1
